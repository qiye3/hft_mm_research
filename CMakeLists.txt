cmake_minimum_required(VERSION 3.20)
project(hft_mm_research VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# 上游独立验收使用 build/upstream；主项目只链接其公开 targets。
set(HBT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(HBT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/hftbacktest-cpp)
add_library(mm_observer src/market_observer.cpp src/replay.cpp)
target_include_directories(mm_observer PUBLIC include)
target_link_libraries(mm_observer PUBLIC hbt::hbt hbt::io)
target_compile_options(mm_observer PRIVATE -Wall -Wextra -Wpedantic)
add_executable(replay_probe src/replay_probe.cpp)
target_link_libraries(replay_probe PRIVATE mm_observer)
add_executable(benchmark_replay benchmarks/benchmark_replay.cpp)
target_link_libraries(benchmark_replay PRIVATE mm_observer)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_market_observer tests/test_market_observer.cpp)
    target_link_libraries(test_market_observer PRIVATE mm_observer)
    add_test(NAME mm.market_observer COMMAND test_market_observer)
endif()
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME mm.tardis_conversion
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_convert_tardis.py $<TARGET_FILE:replay_probe>)
endif()
