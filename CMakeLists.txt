cmake_minimum_required(VERSION 3.20)
project(hft_mm_research VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# 上游独立验收位于 workspace/build/upstream。主项目链接公开 targets。
set(HBT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(HBT_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/hftbacktest-cpp)
add_library(mm_observer src/mm/market_observer.cpp src/mm/replay.cpp)
target_include_directories(mm_observer PUBLIC src)
target_link_libraries(mm_observer PUBLIC hbt::hbt hbt::io)
target_compile_options(mm_observer PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(mm_observer PRIVATE MM_PROJECT_ROOT="${CMAKE_SOURCE_DIR}")
add_executable(replay_probe src/apps/replay_probe.cpp)
target_link_libraries(replay_probe PRIVATE mm_observer)
add_executable(benchmark_replay tests/benchmarks/benchmark_replay.cpp)
target_link_libraries(benchmark_replay PRIVATE mm_observer)
add_library(mm_research src/mm/research_sampler.cpp)
target_link_libraries(mm_research PUBLIC mm_observer)
target_compile_options(mm_research PRIVATE -Wall -Wextra -Wpedantic)
add_executable(extract_research src/apps/extract_research.cpp)
target_link_libraries(extract_research PRIVATE mm_research)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_market_observer tests/test_market_observer.cpp)
    target_link_libraries(test_market_observer PRIVATE mm_observer)
    add_test(NAME mm.market_observer COMMAND test_market_observer)
endif()
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME mm.tardis_conversion
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_convert_tardis.py $<TARGET_FILE:replay_probe>)
    add_test(NAME mm.research_dataset
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_research_dataset.py $<TARGET_FILE:extract_research>)
    add_test(NAME mm.research_versions
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_research_versions.py)
    add_test(NAME mm.feature_examples
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_feature_examples.py)
    add_test(NAME mm.standard_data
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_standard_data.py)
    add_test(NAME mm.storage
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_storage.py)
    add_test(NAME mm.factors
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_factors.py)
    add_test(NAME mm.evaluation
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_evaluation.py)
endif()

add_library(mm_strategy src/mm/market_maker.cpp src/mm/trade_recorder.cpp)
target_include_directories(mm_strategy PUBLIC src)
target_link_libraries(mm_strategy PUBLIC hbt::hbt hbt::io)
target_compile_options(mm_strategy PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(mm_strategy PRIVATE MM_PROJECT_ROOT="${CMAKE_SOURCE_DIR}")
add_executable(baseline_mm src/apps/baseline_mm.cpp)
target_link_libraries(baseline_mm PRIVATE mm_strategy)
if(BUILD_TESTING)
    add_executable(test_market_maker tests/test_market_maker.cpp)
    target_include_directories(test_market_maker PRIVATE src)
    target_link_libraries(test_market_maker PRIVATE mm_strategy)
    add_test(NAME mm.market_maker COMMAND test_market_maker)
endif()
